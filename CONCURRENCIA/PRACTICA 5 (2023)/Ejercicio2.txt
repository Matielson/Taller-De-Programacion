programa Ejercicio2Practica5

procesos
  proceso juntarFlores (ES cant: numero)
  comenzar
    mientras HayFlorEnLaEsquina
      tomarFlor
      cant:= cant + 1
  fin
  
  proceso depositarEsq50 (E cant: numero)
  variables
    avOri, caOri: numero
  comenzar
    avOri:= PosAv
    caOri:= PosCa
    BloquearEsquina(50,50)
    Pos(50,50)
    repetir cant
      depositarFlor
    Pos(avOri, caOri)
    LiberarEsquina(50,50)
  fin
  
  proceso EnviarMensajes(E ID: numero)
  comenzar
    si (ID = 1)
      EnviarMensaje(V, r2)
      EnviarMensaje(V, r3)
    sino
      si (ID = 2)
        EnviarMensaje(V, r1)
        EnviarMensaje(V, r3)
      sino
        EnviarMensaje(V, r1)
        EnviarMensaje(V, r2)
  fin
  
  proceso RecibirMensajes
  variables
    seguir: boolean
  comenzar
    RecibirMensaje(seguir, *)
    RecibirMensaje(seguir, *)
  fin

areas
  areaRecolector1: AreaP (1,1,100,1)
  areaRecolector2: AreaP (1,2,100,2)
  areaRecolector3: AreaP (1,3,100,3)
  areaFiscalizador: AreaP (1,4,1,4)
  areaDeposito: AreaC (50,50,50,50)

robots
  robot recolector
  variables
    ID: numero
    cantEsq: numero
    cantFlo: numero
  comenzar
    RecibirMensaje(ID, rFisc)
    derecha
    mientras (PosAv <> 100)
      cantFlo:= 0
      cantEsq:= 0
      mientras (PosAv <> 100) & (cantEsq < 10)
        juntarFlores(cantFlo)
        cantEsq:= cantEsq + 1
        mover
      si (PosAv = 100)
        juntarFlores(cantFlo)
      depositarEsq50(cantFlo)
      EnviarMensajes(ID)
      RecibirMensajes
    si (ID = 1)
      EnviarMensaje(V, rFisc)
    sino
      si (ID = 2)
        EnviarMensaje(V, rFisc)
      sino
        EnviarMensaje(V, rFisc)
  fin
  
  robot fiscalizador
  variables
    cant: numero
    seguir: boolean
  comenzar
    EnviarMensaje(1, r1)
    EnviarMensaje(2, r2)
    EnviarMensaje(3, r3)
    
    RecibirMensaje(seguir, *)
    RecibirMensaje(seguir, *)
    RecibirMensaje(seguir, *)
    
    Pos(50,50)
    cant:= 0
    mientras HayFlorEnLaEsquina
      tomarFlor
      cant:= cant + 1
    Informar(cant)
  fin

variables
  r1: recolector
  r2: recolector
  r3: recolector
  rFisc: fiscalizador
comenzar
  AsignarArea(r1, areaRecolector1)
  AsignarArea(r2, areaRecolector2)
  AsignarArea(r3, areaRecolector3)
  AsignarArea(rFisc, areaFiscalizador)
  
  AsignarArea(r1, areaDeposito)
  AsignarArea(r2, areaDeposito)
  AsignarArea(r3, areaDeposito)
  AsignarArea(rFisc, areaDeposito)
  
  Iniciar(r1, 1,1)
  Iniciar(r2, 1,2)
  Iniciar(r3, 1,3)
  Iniciar(rFisc, 1,4)
fin
