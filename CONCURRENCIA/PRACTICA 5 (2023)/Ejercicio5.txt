programa Ejercicio5Pracitca5

procesos
  proceso recolectarDeAUno(E num: numero)
  variables
    avOri, caOri: numero
  comenzar
    avOri:= PosAv
    caOri:= PosCa
    repetir num
      BloquearEsquina(100,1)
      Pos(100,1)
      tomarFlor
      Pos(avOri, caOri)
      LiberarEsquina(100,1)
      depositarFlor
  fin
  
  proceso EnviarMensajeCant (E ID: numero; E cant: numero)
  comenzar
    si (ID = 1)
      EnviarMensaje(cant, r1)
    sino
      si (ID = 2)
        EnviarMensaje(cant, r2)
      sino
        EnviarMensaje(cant, r3)
  fin
  
areas
  areaRobotCliente1: AreaP (10,1,10,1)
  areaRobotCliente2: AreaP (11,1,11,1)
  areaRobotCliente3: AreaP (12,1,12,1)
  
  areaRobotServidor: AreaP (13,1,13,1)
  
  areaDeposito: AreaC (100,1,100,1)
  
robots

  robot cliente
  variables
    cantARecolectar: numero
    ID: numero
  comenzar
    RecibirMensaje(ID, rServ)
    repetir 4
      EnviarMensaje(ID, rServ)
      RecibirMensaje(cantARecolectar, rServ)
      recolectarDeAUno(cantARecolectar)
    EnviarMensaje(0, rServ)
  fin
  
  robot servidor
  variables
    cantRobotsTerminados: numero
    avOri, caOri: numero
    ID: numero
    cant: numero
  comenzar
    EnviarMensaje(1, r1)
    EnviarMensaje(2, r2)
    EnviarMensaje(3, r3)
    cantRobotsTerminados:= 0
    avOri:= PosAv
    caOri:= PosCa
    mientras (cantRobotsTerminados < 3)
      RecibirMensaje(ID, *)
      si (ID <> 0)
        Random(cant, 1,5)
        BloquearEsquina(100,1)
        Pos(100,1)
        repetir cant
          depositarFlor
        Pos(avOri, caOri)
        LiberarEsquina(100,1)
        EnviarMensajeCant(ID, cant)
      sino
        cantRobotsTerminados:= cantRobotsTerminados + 1
  fin
  
variables
  r1: cliente
  r2: cliente
  r3: cliente
  rServ: servidor
comenzar
  AsignarArea(r1, areaRobotCliente1)
  AsignarArea(r2, areaRobotCliente2)
  AsignarArea(r3, areaRobotCliente3)
  AsignarArea(rServ, areaRobotServidor)
  
  AsignarArea(r1, areaDeposito)
  AsignarArea(r2, areaDeposito)
  AsignarArea(r3, areaDeposito)
  AsignarArea(rServ, areaDeposito)
  
  Iniciar(r1, 10,1)
  Iniciar(r2, 11,1)
  Iniciar(r3, 12,1)
  Iniciar(rServ, 13,1)
fin
