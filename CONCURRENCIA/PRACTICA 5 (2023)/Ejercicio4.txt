programa Ejercicio4Practica5

procesos
  proceso EnviarMensajes(E av: numero; E ca: numero; E numRobot: numero)
  comenzar
    si (numRobot = 1)
      EnviarMensaje(av,r1)
      EnviarMensaje(ca,r1)
    sino
      si (numRobot = 2)
        EnviarMensaje(av,r2)
        EnviarMensaje(ca,r2)
      sino
        EnviarMensaje(av,r3)
        EnviarMensaje(ca,r3)
  fin
  
  proceso limpiarFlores (ES cantFlo: numero)
  comenzar
    mientras (HayFlorEnLaEsquina)
      tomarFlor
      cantFlo:= cantFlo + 1
  fin
  
  proceso EnviarMensajeFinalizar
  comenzar
    EnviarMensaje(0, r1)
    EnviarMensaje(0, r2)
    EnviarMensaje(0, r3)
  fin
  
  proceso depositarCantEsq10 (E av: numero; E ca: numero; E cantFlo: numero)
  comenzar
    BloquearEsquina(10,10)
    Pos(10,10)
    repetir cantFlo
      depositarFlor
    Pos(av,ca)
    LiberarEsquina(10,10)
  fin
  
areas
  areaFlorero1: AreaP (1,1,1,1)
  areaFlorero2: AreaP (2,1,2,1)
  areaFlorero3: AreaP (3,1,3,1)
  areaFiscalizador: AreaP (4,1,4,1)
  areaDeposito: AreaPC (10,10,10,10)
  areaCuadrante: AreaPC (40,40,60,60)
  
robots
  
  robot florero
  variables
    ID: numero
    av, ca, avOri, caOri: numero
    cantFlo: numero 
  comenzar
    cantFlo:= 0
    avOri:= PosAv
    caOri:= PosCa
    RecibirMensaje(ID, rFisc)
    RecibirMensaje(av, rFisc)
    mientras (av <> 0)
      RecibirMensaje(ca, rFisc)
      
      BloquearEsquina(av,ca)
      Pos(av,ca)
      limpiarFlores(cantFlo)
      Pos(avOri, caOri)
      LiberarEsquina(av,ca)
      
      RecibirMensaje(av, rFisc)
    
    depositarCantEsq10(avOri,caOri,cantFlo)
    EnviarMensaje(cantFlo, rFisc)
  fin

  robot fiscalizador
  variables
    av,ca, cantFlo, cantTotal: numero
    numRobot: numero
  comenzar
    EnviarMensaje(1, r1)
    EnviarMensaje(2, r2)
    EnviarMensaje(3, r3)
    repetir 8
      Random(av,40,60)
      Random(ca,40,60)
      Random(numRobot,1,3)
      EnviarMensajes(av, ca, numRobot)
    EnviarMensajeFinalizar
    
    cantTotal:= 0
    repetir 3
      RecibirMensaje(cantFlo, *)
      cantTotal:= cantTotal + cantFlo
    
    Informar(cantTotal)
  fin
  
variables
  r1: florero
  r2: florero
  r3: florero
  rFisc: fiscalizador
comenzar
  AsignarArea(r1, areaFlorero1)
  AsignarArea(r2, areaFlorero2)
  AsignarArea(r3, areaFlorero3)
  AsignarArea(rFisc, areaFiscalizador)
  
  AsignarArea(r1, areaDeposito)
  AsignarArea(r2, areaDeposito)
  AsignarArea(r3, areaDeposito)
  
  AsignarArea(r1, areaCuadrante)
  AsignarArea(r2, areaCuadrante)
  AsignarArea(r3, areaCuadrante)
  AsignarArea(rFisc, areaCuadrante)
  
  Iniciar(r1, 1,1)
  Iniciar(r2, 2,1)
  Iniciar(r3, 3,1)
  Iniciar(rFisc, 4,1)
fin
