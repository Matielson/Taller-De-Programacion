programa Ejercicio3Practica1
procesos
  
  proceso limpiarEsquina(ES pap: numero; ES flo: numero)
  comenzar
    mientras HayFlorEnLaEsquina
      tomarFlor
      flo:= flo + 1
    mientras HayPapelEnLaEsquina
      tomarPapel
      pap:= pap + 1
  fin
  
  proceso izquierda
  comenzar
    repetir 2
      derecha
  fin

  proceso hacerEscalon(E medida: numero; ES supero: boolean)
  variables
    pap: numero
    flo: numero
  comenzar
    pap:= 0
    flo:= 0
    repetir 2
      repetir medida
        limpiarEsquina(pap, flo)
        mover
      derecha
    si pap = (flo + 1)
      supero:= V
  fin
  
areas
  ciudad: AreaC (1,1,100,100)

robots
  robot robot1
  variables
    medida: numero
    supero: boolean
    cantEsc: numero
  comenzar
    medida:= 1
    cantEsc:= 0
    repetir 4
      supero:= F
      hacerEscalon(medida, supero)
      izquierda
      si (supero = V)
        cantEsc:= cantEsc + 1
      medida:= medida + 1
    Informar(cantEsc)
  fin
