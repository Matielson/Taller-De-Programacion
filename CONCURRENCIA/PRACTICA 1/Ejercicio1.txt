  proceso recorrerAvenida(ES flores: numero; ES esq: numero)
  comenzar
    repetir 99
      mover
      si HayFlorEnLaEsquina
        mientras HayFlorEnLaEsquina
          tomarFlor
          flores:= flores + 1
      sino
        esq:= esq + 1
    repetir flores
      depositarFlor
  fin

    flores: numero
    esquinas: numero
  comenzar
    recorrerAvenida(flores, esquinas)
    Informar(flores)
    Informar(esquinas)
  fin
