  robot recolector
  variables
    ID: numero
    aux: boolean
    objetos: numero
    avActual, caActual, av, ca: numero
  comenzar
    RecibirMensaje(ID, rFisc)
    RecibirMensaje(aux, rFisc)
    avActual:= PosAv
    caActual:= PosCa
    
    mientras aux = V
      objetos:= 0
      Random(av, 25,75)
      Random(ca, 25,75)
      Pos(av,ca)
      limpiarEsq(objetos)
      Pos(avActual, caActual)
      EnviarMensaje(objetos, rFisc)
      RecibirMensaje(aux, rFisc)
      
    RecibirMensaje(aux, rFisc)
    si aux = V
      Pos(50,50)
  fin
  
  robot fiscalizador
  variables
    num: numero
    cantR1, cantR2, cantR3, cantR4: numero
    max, IDmax: numero
    objetos: numero
  comenzar
    EnviarMensaje(1, r1)
    EnviarMensaje(2, r2)
    EnviarMensaje(3, r3)
    EnviarMensaje(4, r4)
    cantR1:= 0
    cantR2:= 0
    cantR3:= 0
    cantR4:= 0
        
    repetir 10
      Random(num,1,4)  
      si num = 1
        EnviarMensaje(V, r1)
        RecibirMensaje(objetos, r1)
        cantR1:= cantR1 + objetos
      sino
        si num = 2
          EnviarMensaje(V, r2)
          RecibirMensaje(objetos, r2)
          cantR2:= cantR2 + objetos
        sino
          si num = 3
            EnviarMensaje(V, r3)
            RecibirMensaje(objetos, r3)
            cantR3:= cantR3 + objetos
          sino
            EnviarMensaje(V, r4)
            RecibirMensaje(objetos, r4)
            cantR4:= cantR4 + objetos
      
    max:= cantR1
    IDmax:= 1
    si cantR2 > max
      max:= cantR2
      IDmax:= 2
    si cantR3 > max
      max:= cantR3
      IDmax:= 3
    si cantR4 > max
      max:= cantR4
      IDmax:= 4
        
    EnviarMensaje(F, r1)
    EnviarMensaje(F, r2)
    EnviarMensaje(F, r3)
    EnviarMensaje(F, r4)
      
    si IDmax = 1
      EnviarMensaje(V, r1)
      EnviarMensaje(F, r2)
      EnviarMensaje(F, r3)
      EnviarMensaje(F, r4)
    sino
      si IDmax = 2
        EnviarMensaje(F, r1)
        EnviarMensaje(V, r2)
        EnviarMensaje(F, r3)
        EnviarMensaje(F, r4)
      sino
        si IDmax = 3
          EnviarMensaje(F, r1)
          EnviarMensaje(F, r2)
          EnviarMensaje(V, r3)
          EnviarMensaje(F, r4)  
        sino
          EnviarMensaje(F, r1)
          EnviarMensaje(F, r2)
          EnviarMensaje(F, r3)
          EnviarMensaje(V, r4)                      
  fin
