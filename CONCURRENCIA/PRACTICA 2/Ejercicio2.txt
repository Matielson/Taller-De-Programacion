procesos

  proceso limpiarEsq(ES pap: numero; ES flo: numero)
  comenzar
    mientras HayPapelEnLaEsquina
      tomarPapel
      pap:= pap + 1
    mientras HayFlorEnLaEsquina
      tomarFlor
      flo:= flo + 1
  fin

  proceso hacerEscalon(E alto: numero; ES pap: numero; ES flo: numero)
  comenzar
    repetir alto
      limpiarEsq(pap,flo)
      mover
    derecha
    limpiarEsq(pap,flo)
    mover
  fin
  
  proceso izquierda
  comenzar
    repetir 3
      derecha
  fin
  
  proceso verificarEscalon (E pap: numero; E flo: numero; ES cant: numero)
  comenzar
    si (flo > pap)
      cant:= cant + 1
  fin

areas
  ciudad: AreaC (1,1,100,100)

robots
  robot robotJuntador
  variables
    pap, flo: numero
    alto: numero
    cant: numero
  comenzar
    cant:= 0
    repetir 3
      pap:= 0
      flo:= 0
      Random(alto, 1,5)
      hacerEscalon(alto, pap, flo)
      izquierda
      verificarEscalon(pap, flo, cant)
      
    pap:= 0
    flo:= 0
    Random(alto, 1,5)
    hacerEscalon(alto,pap,flo)
    limpiarEsq(pap,flo)
    verificarEscalon(pap, flo, cant)
    
    EnviarMensaje(cant, rJefe)
  fin
  
  robot jefe
  variables
    cant, total: numero
  comenzar
    total:= 0
    repetir 3
      RecibirMensaje(cant, *)
      total:= total + cant
    Informar(total)
  fin
