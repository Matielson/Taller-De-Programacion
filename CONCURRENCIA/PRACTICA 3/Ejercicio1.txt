programa ejercicio1Practica3

procesos

  proceso limpiarFlores(ES flo: numero)
  comenzar
    mientras HayFlorEnLaEsquina
      tomarFlor
      flo:= flo + 1
  fin

  proceso intentoFlorero (ES flo: numero)
  variables
    av,ca,avActual, caActual: numero
  comenzar
    avActual:= PosAv
    caActual:= PosCa
    Random(av,1,5)
    Random(ca,1,10)
    BloquearEsquina(av,ca)
    Pos(av,ca)
    limpiarFlores(flo)
    Pos(avActual, caActual)
    LiberarEsquina(av,ca)
  fin
  
  proceso limpiarPapeles(ES pap: numero)
  comenzar
    mientras HayPapelEnLaEsquina
      tomarPapel
      pap:= pap + 1
  fin
  
  proceso intentoPapelero (ES pap: numero)
  variables
    av,ca,avActual, caActual: numero
  comenzar
    avActual:= PosAv
    caActual:= PosCa
    Random(av,6,10)
    Random(ca,1,9)
    BloquearEsquina(av,ca)
    Pos(av,ca)
    limpiarPapeles(pap)
    Pos(avActual, caActual)
    LiberarEsquina(av,ca)
  fin
  
areas
  areaFloreros: AreaPC (1,1,5,10)
  areaPapeleros: AreaPC (6,1,10,9)
  areaInicio1: AreaP (6,10,6,10)
  areaInicio2: AreaP (7,10,7,10)
  areaInicio3: AreaP (8,10,8,10)
  areaInicio4: AreaP (9,10,9,10)
  areaDeposito: AreaC (10,10,10,10)
  
robots

  robot robotFlorero
  variables
    caActual, avActual: numero
    cantFlo: numero
  comenzar
    avActual:= PosAv
    caActual:= PosCa
    cantFlo:= 0
    repetir 5
      intentoFlorero(cantFlo)
    repetir cantFlo
      BloquearEsquina(10,10)
      Pos(10,10)
      depositarFlor
      Pos(avActual, caActual)
      LiberarEsquina(10,10)
  fin
  
  robot robotPapelero
  variables
    caActual, avActual: numero
    cantPap: numero
  comenzar
    avActual:= PosAv
    caActual:= PosCa
    cantPap:= 0
    repetir 5
      intentoPapelero(cantPap)
    repetir cantPap
      BloquearEsquina(10,10)
      Pos(10,10)
      depositarPapel
      Pos(avActual, caActual)
      LiberarEsquina(10,10)
  fin
