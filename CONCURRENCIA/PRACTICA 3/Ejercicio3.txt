programa Ejercicio3Practica3

procesos
  
  proceso agarrarFlorEsq (ES cantFlores: numero; ES vacia: boolean)
  comenzar
    si HayFlorEnLaEsquina
      tomarFlor
    
    si ~HayFlorEnLaEsquina
      vacia:= V
  fin

areas
  cuadrante: AreaPC (2,2,10,10)
  areaJefe: AreaP (1,1,1,1)
  areaR1: AreaP(2,1,2,1)
  areaR2: AreaP(3,1,3,1)
  areaR3: AreaP(4,1,4,1)
  areaR4: AreaP(5,1,5,1)  
  
robots
  robot recolector
  variables
    av, ca, avOri, caOri: numero
    vacia: boolean
    cantFlores: numero
    ID: numero
  comenzar
    RecibirMensaje(ID, rJefe)
    vacia:= F
    avOri:= PosAv
    caOri:= PosCa
    cantFlores:= 0
    RecibirMensaje(av, rJefe)
    RecibirMensaje(ca, rJefe)
    
    mientras (vacia = F)
      BloquearEsquina(av,ca)
      Pos(av,ca)
      agarrarFlorEsq(cantFlores, vacia)
      Pos(avOri, caOri)
      LiberarEsquina(av,ca)
      
    EnviarMensaje(ID, rJefe)
    EnviarMensaje(cantFlores, rJefe)
  fin
  
  robot jefe
  variables
    av,ca: numero
    cant, cantMax, ID, IDmax: numero
  comenzar
    Random(av, 2,10)
    Random(ca, 2,10)
    
    EnviarMensaje(1, r1)
    EnviarMensaje(2, r2)
    EnviarMensaje(3, r3)
    EnviarMensaje(4, r4)
    
    EnviarMensaje(av, r1)
    EnviarMensaje(ca, r1)
    EnviarMensaje(av, r2)
    EnviarMensaje(ca, r2)
    EnviarMensaje(av, r3)
    EnviarMensaje(ca, r3)
    EnviarMensaje(av, r4)
    EnviarMensaje(ca, r4)
    
    cantMax:= -1
    
    repetir 4
      RecibirMensaje(ID, *)
      si (ID = 1)
        RecibirMensaje(cant, r1)
      sino
        si (ID = 2)
          RecibirMensaje(cant, r2)
        sino
          si (ID = 3)
            RecibirMensaje(cant, r3)
          sino
            RecibirMensaje(cant, r4)
            
      si cant > cantMax
        cantMax:= cant
        IDmax:= ID
    
    Informar(IDmax)
  fin

variables
  r1: recolector
  r2: recolector
  r3: recolector
  r4: recolector
  rJefe: jefe
comenzar
  AsignarArea(r1, areaR1)
  AsignarArea(r2, areaR2)
  AsignarArea(r3, areaR3)
  AsignarArea(r4, areaR4)
  AsignarArea(rJefe, areaJefe)
  
  AsignarArea(r1, cuadrante) 
  AsignarArea(r2, cuadrante)
  AsignarArea(r3, cuadrante)
  AsignarArea(r4, cuadrante)
  AsignarArea(rJefe, cuadrante) 
    
  Iniciar(r1, 2,1)
  Iniciar(r2, 3,1)
  Iniciar(r3, 4,1)
  Iniciar(r4, 5,1)
  Iniciar(rJefe, 1,1)
  
fin
