procesos
  proceso tomarFlorEsq10(ES vacio: boolean)
  variables
    avOri, caOri: numero
  comenzar
    avOri:= PosAv
    caOri:= PosCa
    BloquearEsquina(10,10)
    Pos(10,10)
    si HayFlorEnLaEsquina
      tomarFlor
      si ~HayFlorEnLaEsquina
        vacio:= V
      BloquearEsquina(11,11)
      Pos(11,11)
      LiberarEsquina(10,10)
      depositarFlor
      Pos(avOri, caOri)
      LiberarEsquina(11,11)
    sino
      vacio:= V
      Pos(avOri, caOri)
      LiberarEsquina(10,10)
  fin
  
  proceso limpiarEsq11
  comenzar
    Pos(11,11)
    mientras HayFlorEnLaEsquina

  fin
