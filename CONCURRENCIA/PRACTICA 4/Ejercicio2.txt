programa Ejercicio2Practica4

procesos

  proceso depositarPapelesEsq
  variables
    avOri, caOri: numero
  comenzar
    avOri:= PosAv
    caOri:= PosCa
    BloquearEsquina(50,50)
    Pos(50,50)
    repetir 5
      depositarPapel
    Pos(avOri, caOri)
    LiberarEsquina(50,50)
  fin
  
  proceso depositarTodo
  comenzar
    mientras HayPapelEnLaEsquina
      depositarPapel
  fin
  
  proceso intentarTomarPapeles(E cant: numero; ES pudo: boolean)
  variables
    avOri, caOri: numero
    cantAux: numero
  comenzar
    avOri:= PosAv
    caOri:= PosCa
    cantAux:= 0
    BloquearEsquina(50,50)
    Pos(50,50)
    mientras ((HayPapelEnLaEsquina) & (cantAux < cant))
      tomarPapel
      cantAux:= cantAux + 1
    si (cantAux = cant)
      Pos(avOri, caOri)
      LiberarEsquina(50,50)
      depositarTodo
      pudo:= V
    sino
      depositarTodo
      Pos(avOri, caOri)
      LiberarEsquina(50,50)
      pudo:= F
  fin
  
areas
  areaProductor1: AreaP (5,1,5,100)
  areaProductor2: AreaP (10,1,10,100)
  areaConsumidor1: AreaP (11,1,11,1)
  areaConsumidor2: AreaP (12,1,12,1)
  areaDeposito: AreaC (50,50,50,50)
  
robots
  robot productor
  variables
    cantPapeles: numero
  comenzar
    cantPapeles:= 0
    mientras (PosCa <> 100)
      mientras ((HayPapelEnLaEsquina) & (cantPapeles < 5))
        tomarPapel
        cantPapeles:= cantPapeles + 1
      
      si (cantPapeles = 5)
        depositarPapelesEsq
        cantPapeles:= 0
      sino
        mover
  fin
  
  robot consumidor
  variables
    cant: numero
    intentos: numero
    pudo: boolean
  comenzar
    Random(cant,2,5)
    intentos:= 0
    mientras (intentos < 8)
      intentarTomarPapeles(cant, pudo)
      si (pudo = F)
        intentos:= intentos + 1
      sino
        intentos:= 0
  fin
  
variables
  rProd1: productor

comenzar
  AsignarArea(rProd1, areaProductor1)

  AsignarArea(rCons2, areaConsumidor2)
  
  AsignarArea(rProd1, areaDeposito)
  AsignarArea(rProd2, areaDeposito)
  AsignarArea(rCons1, areaDeposito)
  AsignarArea(rCons2, areaDeposito)
  
  Iniciar(rProd1, 5,1)
  Iniciar(rProd2, 10,1)

  Iniciar(rCons2, 12,1)
fin
