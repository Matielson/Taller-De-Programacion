programa Ejercicio4Practica4

procesos
  proceso recogerFlores
  comenzar
    mientras HayFlorEnLaEsquina
      tomarFlor
  fin
  
  proceso recogerPapeles
  comenzar
    mientras HayPapelEnLaEsquina
      tomarPapel
  fin
  
  proceso vaciarBolsa
  comenzar
    mientras HayFlorEnLaBolsa
      depositarFlor
    mientras HayPapelEnLaBolsa
      depositarPapel
  fin
  
  proceso hacerTarea (E num: numero)
  comenzar
    si (num = 1)
      recogerFlores
    sino
      si (num = 2)
        recogerPapeles
      sino
        vaciarBolsa
  fin
  
areas
  areaJefe: AreaP (1,1,1,1)
  areaTrabajador1: AreaP (2,1,2,1)
  areaTrabajador2: AreaP (3,1,3,1)
  areaDeTrabajo: AreaPC (2,2,100,100)

robots
  
  robot trabajador
  variables
    av, ca, avOri, caOri: numero
    tarea: numero
  comenzar
    avOri:= PosAv
    caOri:= PosCa
    RecibirMensaje(tarea, rJefe)
    mientras (tarea <> 4)
      Random(av, 2,100)
      Random(ca, 2,100)
      BloquearEsquina(av, ca)
      Pos(av, ca)
      hacerTarea(tarea)
      Pos(avOri, caOri)
      LiberarEsquina(av,ca)
      RecibirMensaje(tarea, rJefe)
  fin

  robot jefe
  variables
    tarea: numero
    r: numero
  comenzar
    repetir 10
      Random(tarea, 1,3)
      Random(r, 1,2)
      si (r = 1)
        EnviarMensaje(tarea, r1)
      sino
        EnviarMensaje(tarea, r2)
    EnviarMensaje(4, r1)
    EnviarMensaje(4, r2)
  fin
  
variables
  r1: trabajador
  r2: trabajador
  rJefe: jefe
comenzar
  AsignarArea(r1, areaTrabajador1)
  AsignarArea(r2, areaTrabajador2)
  AsignarArea(rJefe, areaJefe)
  AsignarArea(r1, areaDeTrabajo)
  AsignarArea(r2, areaDeTrabajo)
  Iniciar(rJefe, 1,1)
  Iniciar(r1, 2,1)
  Iniciar(r2, 3,1)
fin
