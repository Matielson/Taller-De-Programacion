programa Ejercicio1Practica4
  
areas
  ciudad: AreaC (1,1,100,100)
  
robots
  robot cliente
  variables
    ID: numero
    avOri, caOri, caSig: numero
    seguir: boolean
    cantFlores: numero
    
  comenzar
    RecibirMensaje(ID, rServ)
    
    mientras (PosCa < 100)
      avOri:= PosAv
      caOri:= PosCa
      caSig:= PosCa + 1
      
      {CALCULO LA CANTIDAD DE FLORES Y LE ENVIO LOS DATOS AL SERVIDOR}
      Random(cantFlores,1,4)
      EnviarMensaje(ID, rServ)
      EnviarMensaje(cantFlores, rServ)
      EnviarMensaje(avOri, rServ)
      EnviarMensaje(caSig, rServ)
      
      {ESPERO LA CONFIRMACION PARA CONTINUAR}
      RecibirMensaje(seguir, rServ)
      
      {LAS VOY A BUSCAR Y ME POSICIONO PARA IR DEJANDO}
      mover
      repetir cantFlores
        tomarFlor
      Pos(avOri, caOri)
      
      {MIENTRAS NO ME CAIGA Y TENGA, DEPOSITO}
      mientras ((PosCa < 100) & (HayFlorEnLaBolsa))
        depositarFlor
        mover
      
    {TERMINO, Y LE MANDO AL SERVIDOR QUE YA TERMINE}
    EnviarMensaje(ID, rServ)
    EnviarMensaje(0, rServ)
  fin
  
  robot servidor
  variables
    cantRobotsTerminados: numero
    ID: numero
    cantFlores: numero
    av, ca: numero
  comenzar
    cantRobotsTerminados:= 0
    EnviarMensaje(1, r1)
    EnviarMensaje(2, r2)
    EnviarMensaje(3, r3)
    
    mientras (cantRobotsTerminados < 3)
      RecibirMensaje(ID, *)
      si (ID = 1)
        RecibirMensaje(cantFlores, r1)
      sino
        si (ID = 2)
          RecibirMensaje(cantFlores, r2)
        sino
          RecibirMensaje(cantFlores, r3)
      
      si (cantFlores <> 0)
        si (ID = 1)
          RecibirMensaje(av, r1)
          RecibirMensaje(ca, r1)
        sino
          si (ID = 2)
            RecibirMensaje(av, r2)
            RecibirMensaje(ca, r2)
          sino
            RecibirMensaje(av, r3)
            RecibirMensaje(ca, r3)
            
        Pos(av,ca)
        repetir cantFlores
          depositarFlor
        Pos(100,100)
        
        si (ID = 1)
          EnviarMensaje(V, r1)
        sino
          si (ID = 2)
            EnviarMensaje(V, r2)
          sino
            EnviarMensaje(V, r3)
      
      sino
        cantRobotsTerminados:= cantRobotsTerminados + 1
  fin
  
variables
  r1: cliente
  r2: cliente
  r3: cliente
  rServ: servidor
comenzar
  AsignarArea(r1, ciudad)
  AsignarArea(r2, ciudad)
  AsignarArea(r3, ciudad)
  AsignarArea(rServ, ciudad)
  Iniciar(r1, 1,1)
  Iniciar(r2, 2,1)
  Iniciar(r3, 3,1)
  Iniciar(rServ, 100,100)
fin
